package coders

import (
	"bytes"
	"fmt"
	"go/format"
	"strings"

	gg "github.com/krateoplatformops/crdgen/v2/internal/codegen"
)

func GenTypes(res *Resource) (dat []byte, err error) {
	buf := bytes.Buffer{}

	dat, err = CodeForSpec(res)
	if err != nil {
		return
	}
	buf.Write(dat)

	gen := gg.New()

	grp := gen.NewGroup().AddLine()
	grp.AddLineComment("+kubebuilder:object:root=true")

	grp.AddLineComment("+k8s:deepcopy-gen:interfaces=k8s.io/apimachinery/pkg/runtime.Object")
	if res.StatusSchema != nil {
		grp.AddLineComment("+kubebuilder:subresource:status")
	}

	grp.AddLineComment(
		`+kubebuilder:printcolumn:name="AGE",type="date",JSONPath=".metadata.creationTimestamp"`)
	if res.Managed {
		grp.AddLineComment(
			`+kubebuilder:printcolumn:name="READY",type="string",JSONPath=".status.conditions[?(@.type=='Ready')].status`)
	}

	if len(res.Categories) > 0 {
		categories := strings.Join(res.Categories, ",")
		grp.AddLineComment(fmt.Sprintf(
			"+kubebuilder:resource:scope=Namespaced,categories={%s}", categories))
	} else {
		grp.AddLineComment("+kubebuilder:resource:scope=Namespaced")
	}

	st := gen.NewGroup().NewStruct(res.Kind)
	st.AddField("", "metav1.TypeMeta", map[string]string{
		"json": ",inline",
	})
	st.AddField("", "metav1.ObjectMeta", map[string]string{
		"json": "metadata",
	})
	st.AddField("Spec", res.Kind+"Spec", map[string]string{
		"json": "spec",
	})

	if res.Managed {
		if res.StatusSchema == nil {
			st.AddField("Status", "*runtime.RawExtension", map[string]string{
				"json": "status,omitempty",
			})
		}
	}

	gen.Write(&buf)

	return format.Source(buf.Bytes())
}
