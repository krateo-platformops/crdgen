package coders

import (
	"bytes"

	gg "github.com/krateoplatformops/crdgen/v2/internal/codegen"
	"github.com/krateoplatformops/crdgen/v2/internal/schemas"
)

var (
	generatedStructs = map[string]bool{}
	generatedEnums   = map[string]bool{}
)

func CodeForSpec(res *Resource) (code []byte, err error) {
	var (
		specSchema   *schemas.Schema
		statusSchema *schemas.Schema
	)

	specSchema, err = schemas.FromJSONReader(bytes.NewReader(res.SpecSchema))
	if err != nil {
		return nil, err
	}

	if res.StatusSchema != nil {
		statusSchema, err = schemas.FromJSONReader(bytes.NewReader(res.StatusSchema))
		if err != nil {
			return nil, err
		}
	}

	gen := gg.New()

	pkgs := gen.NewGroup().AddPackage(res.Version).NewImport().
		AddAlias("k8s.io/apimachinery/pkg/apis/meta/v1", "metav1").
		AddPath("k8s.io/apimachinery/pkg/runtime")

	if statusSchema != nil && res.Managed {
		pkgs.AddAlias("commonv1", "github.com/krateoplatformops/provider-runtime/apis/common/v1")
	}

	buildStruct(gen, res.Kind+"Spec", (*schemas.Type)(specSchema.ObjectAsType), specSchema.Definitions)
	if res.StatusSchema != nil {
		buildStruct(gen, res.Kind+"Status", (*schemas.Type)(statusSchema.ObjectAsType), statusSchema.Definitions)
	}

	buf := bytes.Buffer{}
	gen.Write(&buf)

	return buf.Bytes(), nil
}
